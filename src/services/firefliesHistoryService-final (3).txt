const FIREFLIES_API_URL = 'https://api.fireflies.ai/graphql';
const FIREFLIES_API_KEY = '1283ad0e-0606-47fc-b186-b284ced72b53';

export interface FirefliesParticipant {
  name: string;
  email?: string;
  user_id?: string;
  speakingTime?: number;
  wordsSpoken?: number;
}

export interface FirefliesSummary {
  overview?: string;
  keywords?: string[];
  actionItems?: string[];
  keyPoints?: string[];
  decisions?: string[];
  nextSteps?: string[];
}

export interface FirefliesTranscript {
  id: string;
  title: string;
  date: string;
  duration: number;
  transcript: string;
  summary?: FirefliesSummary;
  participants: FirefliesParticipant[];
  keywords?: string[];
  status: 'completed' | 'processing' | 'pending';
}

export interface FirefliesSearchResult {
  transcripts: FirefliesTranscript[];
  totalCount: number;
  hasMore: boolean;
}

// 🔧 Função base para chamadas GraphQL
async function callFirefliesAPI(query: string, variables: any = {}): Promise<any> {
  try {
    console.log('🔥 Chamando API do Fireflies (versão corrigida)...');
    console.log('📡 Query:', query.substring(0, 200) + '...');
    
    const response = await fetch(FIREFLIES_API_URL, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${FIREFLIES_API_KEY}`
      },
      body: JSON.stringify({
        query,
        variables
      })
    });

    console.log('📡 Status da resposta:', response.status);

    if (!response.ok) {
      const errorText = await response.text();
      console.error('❌ Erro HTTP:', response.status, errorText);
      throw new Error(`HTTP ${response.status}: ${response.statusText}`);
    }

    const data = await response.json();
    console.log('📦 Resposta da API:', data);
    
    if (data.errors) {
      console.error('❌ Erros GraphQL:', data.errors);
      throw new Error(`GraphQL Error: ${data.errors.map((e: any) => e.message).join(', ')}`);
    }

    return data.data;
  } catch (error) {
    console.error('❌ Erro na API do Fireflies:', error);
    throw error;
  }
}

// 📅 Funções de formatação
export function formatFirefliesDate(timestamp: string | number): string {
  const date = new Date(typeof timestamp === 'string' ? parseInt(timestamp) : timestamp);
  return date.toLocaleString('pt-BR', {
    year: 'numeric',
    month: 'long',
    day: 'numeric',
    hour: '2-digit',
    minute: '2-digit'
  });
}

export function formatFirefliesDuration(seconds: number): string {
  const hours = Math.floor(seconds / 3600);
  const minutes = Math.floor((seconds % 3600) / 60);
  const secs = seconds % 60;
  
  if (hours > 0) {
    return `${hours}h ${minutes}m`;
  } else if (minutes > 0) {
    return `${minutes}m ${secs}s`;
  } else {
    return `${secs}s`;
  }
}

// 🔍 Buscar reuniões por palavra-chave
export async function searchFirefliesTranscriptions(
  searchTerm: string,
  limit: number = 20
): Promise<FirefliesSearchResult> {
  try {
    // Buscar todas as reuniões e filtrar localmente
    const allMeetings = await getFirefliesTranscriptions(50);
    
    const filteredMeetings = allMeetings.transcripts.filter(meeting => {
      const searchText = `${meeting.title} ${meeting.summary?.overview || ''} ${meeting.transcript}`.toLowerCase();
      return searchText.includes(searchTerm.toLowerCase());
    });

    return {
      transcripts: filteredMeetings.slice(0, limit),
      totalCount: filteredMeetings.length,
      hasMore: filteredMeetings.length > limit
    };
  } catch (error) {
    console.error('❌ Erro na busca:', error);
    return { transcripts: [], totalCount: 0, hasMore: false };
  }
}
