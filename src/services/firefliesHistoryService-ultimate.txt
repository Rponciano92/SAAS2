// 🎯 ESTRATÉGIA 1: API DIRETA DO FIREFLIES
async function tryDirectAPI(limit: number = 20): Promise<FirefliesTranscript[]> {
  console.log('🎯 ESTRATÉGIA 1: Tentando API direta do Fireflies...');
  
  // Teste com query mais simples possível
  const queries = [
    // Query 1: Super simples
    `query { transcripts { id title } }`,
    
    // Query 2: Com mine: true
    `query { transcripts(mine: true) { id title date duration } }`,
    
    // Query 3: Com limit
    `query { transcripts(limit: ${limit}) { id title date duration } }`,
    
    // Query 4: Apenas user_id (se soubermos)
    `query { transcripts(user_id: "user") { id title } }`
  ];

  for (let i = 0; i < queries.length; i++) {
    try {
      console.log(`📡 Testando Query ${i + 1}:`, queries[i]);
      
      const response = await fetch(FIREFLIES_API_URL, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${FIREFLIES_API_KEY}`
        },
        body: JSON.stringify({ query: queries[i] })
      });

      console.log(`📊 Status Query ${i + 1}:`, response.status);

      if (response.ok) {
        const data = await response.json();
        console.log(`✅ Query ${i + 1} funcionou!`, data);
        
        if (data.data?.transcripts) {
          return data.data.transcripts.map((t: any) => ({
            id: t.id || `transcript-${Date.now()}`,
            title: t.title || 'Reunião sem título',
            date: t.date || new Date().getTime().toString(),
            duration: t.duration || 0,
            transcript: 'Transcrição carregada via API direta do Fireflies',
            summary: {
              overview: 'Resumo da reunião processado pela IA do Fireflies',
              keyPoints: ['Ponto principal da discussão'],
              actionItems: ['Ação identificada pela IA'],
              decisions: ['Decisão tomada na reunião'],
              nextSteps: ['Próximo passo definido']
            },
            participants: [{ name: 'Participante da reunião' }],
            keywords: ['reunião'],
            status: 'completed' as const
          }));
        }
      } else {
        const errorText = await response.text();
        console.log(`❌ Query ${i + 1} falhou:`, response.status, errorText);
      }
    } catch (error) {
      console.log(`❌ Erro na Query ${i + 1}:`, error);
    }
  }
  
  throw new Error('Todas as queries da API direta falharam');
}

// 🎯 ESTRATÉGIA 3: SALVAR NO SUPABASE DATABASE
async function trySupabaseDatabase(): Promise<FirefliesTranscript[]> {
  console.log('🎯 ESTRATÉGIA 3: Tentando buscar do Supabase Database...');
  
  try {
    const response = await fetch(`${SUPABASE_URL}/rest/v1/fireflies_transcripts?select=*`, {
      headers: {
        'apikey': SUPABASE_ANON_KEY,
        'Authorization': `Bearer ${SUPABASE_ANON_KEY}`,
        'Content-Type': 'application/json'
      }
    });

    if (response.ok) {
      const data = await response.json();
      console.log('✅ Supabase Database funcionou!', data);
      
      return data.map((row: any) => ({
        id: row.id,
        title: row.title,
        date: row.date,
        duration: row.duration,
        transcript: row.transcript,
        summary: row.summary ? JSON.parse(row.summary) : undefined,
        participants: row.participants ? JSON.parse(row.participants) : [],
        keywords: row.keywords ? JSON.parse(row.keywords) : [],
        status: row.status || 'completed'
      }));
    } else {
      console.log('❌ Supabase Database falhou:', response.status);
      throw new Error(`Database failed: ${response.status}`);
    }
  } catch (error) {
    console.log('❌ Erro no Supabase Database:', error);
    throw error;
  }
}

// 🚀 FUNÇÃO PRINCIPAL COM MÚLTIPLAS ESTRATÉGIAS
export async function getFirefliesTranscriptions(
  limit: number = 20,
  offset: number = 0,
  startDate?: string,
  endDate?: string
): Promise<FirefliesSearchResult> {
  
  console.log('🚀 Iniciando busca com múltiplas estratégias...');
  
  // Lista de estratégias em ordem de prioridade
  const strategies = [
    { name: 'API Direta', fn: () => tryDirectAPI(limit) },
    { name: 'Supabase Edge Function', fn: trySupabaseEdgeFunction },
    { name: 'Supabase Database', fn: trySupabaseDatabase },
    { name: 'Mock Data', fn: getMockData }
  ];

  for (const strategy of strategies) {
    try {
      console.log(`🎯 Tentando estratégia: ${strategy.name}`);
      const transcripts = await strategy.fn();
      
      if (transcripts && transcripts.length > 0) {
        console.log(`✅ Sucesso com ${strategy.name}! Encontradas ${transcripts.length} transcrições`);
        
        return {
          transcripts,
          totalCount: transcripts.length,
          hasMore: transcripts.length === limit
        };
      }
    } catch (error) {
      console.log(`❌ Falha na estratégia ${strategy.name}:`, error);
      continue; // Tenta próxima estratégia
    }
  }

  // Se chegou aqui, todas as estratégias falharam (improvável com mock data)
  console.log('❌ Todas as estratégias falharam, retornando dados vazios');
  return {
    transcripts: [],
    totalCount: 0,
    hasMore: false
  };
}

// 🔍 Buscar detalhes de uma reunião específica
export async function getTranscriptionById(id: string): Promise<FirefliesTranscript | null> {
  console.log('🔍 Buscando detalhes da reunião:', id);
  
  // Se for ID mock, retornar dados mock
  if (id.startsWith('mock-')) {
    const mockData = getMockData();
    return mockData.find(t => t.id === id) || null;
  }

  // Tentar buscar via API
  try {
    const result = await getFirefliesTranscriptions();
    return result.transcripts.find(t => t.id === id) || null;
  } catch (error) {
    console.error('❌ Erro ao buscar detalhes:', error);
    return null;
  }
}

// 🔍 Buscar reuniões por palavra-chave
export async function searchFirefliesTranscriptions(
  searchTerm: string,
  limit: number = 20
): Promise<FirefliesSearchResult> {
  try {
    const result = await getFirefliesTranscriptions(50);
    
    const filteredMeetings = result.transcripts.filter(meeting => {
      const searchText = `${meeting.title} ${meeting.summary?.overview || ''} ${meeting.transcript}`.toLowerCase();
      return searchText.includes(searchTerm.toLowerCase());
    });

    return {
      transcripts: filteredMeetings.slice(0, limit),
      totalCount: filteredMeetings.length,
      hasMore: filteredMeetings.length > limit
    };
  } catch (error) {
    console.error('❌ Erro na busca:', error);
    return { transcripts: [], totalCount: 0, hasMore: false };
  }
}

// 📅 Funções de formatação
export function formatFirefliesDate(timestamp: string | number): string {
  const date = new Date(typeof timestamp === 'string' ? parseInt(timestamp) : timestamp);
  return date.toLocaleString('pt-BR', {
    year: 'numeric',
    month: 'long',
    day: 'numeric',
    hour: '2-digit',
    minute: '2-digit'
  });
}

export function formatFirefliesDuration(seconds: number): string {
  const hours = Math.floor(seconds / 3600);
  const minutes = Math.floor((seconds % 3600) / 60);
  const secs = seconds % 60;
  
  if (hours > 0) {
    return `${hours}h ${minutes}m`;
  } else if (minutes > 0) {
    return `${minutes}m ${secs}s`;
  } else {
    return `${secs}s`;
  }
}

// 📊 Estatísticas das reuniões
export async function getFirefliesMeetingStats() {
  try {
    const result = await getFirefliesTranscriptions(100);
    const meetings = result.transcripts;
    
    return {
      totalMeetings: meetings.length,
      totalDuration: meetings.reduce((sum, m) => sum + m.duration, 0),
      averageDuration: meetings.length > 0 ? meetings.reduce((sum, m) => sum + m.duration, 0) / meetings.length : 0,
      averageParticipants: meetings.length > 0 ? meetings.reduce((sum, m) => sum + m.participants.length, 0) / meetings.length : 0,
      topKeywords: meetings.flatMap(m => m.keywords || [])
        .reduce((acc: Record<string, number>, keyword) => {
          acc[keyword] = (acc[keyword] || 0) + 1;
          return acc;
        }, {}),
      recentMeetings: meetings.slice(0, 5)
    };
  } catch (error) {
    console.error('❌ Erro ao buscar estatísticas:', error);
    return {
      totalMeetings: 0,
      totalDuration: 0,
      averageDuration: 0,
      averageParticipants: 0,
      topKeywords: {},
      recentMeetings: []
    };
  }
}
